// --- 🚀 ฟังก์ชันดึง Feed (อัปเกรดใหม่ 5 ฟีเจอร์) --- function fetchFeed() { const container = document.getElementById('feedContainer'); const filterType = currentFeedFilter; // all, related, request const filterDate = document.getElementById('filterDate').value; const filterYear = document.getElementById('filterYear').value; if (!container) return; container.innerHTML = '
'; fetch(GAS_URL + '?action=get_feed&t=' + Date.now()) .then(r => r.json()) .then(feed => { container.innerHTML = ''; // 🔔 แจ้งเตือนใหม่ (Badge) const lastSeen = parseInt(localStorage.getItem('lastSeenStoryCount') || 0); const newStories = feed.length - lastSeen; const badgeBox = document.getElementById('nav-stories-btn'); // ล้างจุดแดงเก่า if(badgeBox.querySelector('.nav-notify-badge')) badgeBox.querySelector('.nav-notify-badge').remove(); if (newStories > 0 && !document.getElementById('page-stories').classList.contains('active')) { badgeBox.innerHTML += `
${newStories}
`; } else if (document.getElementById('page-stories').classList.contains('active')) { localStorage.setItem('lastSeenStoryCount', feed.length); } // 🔍 กรองข้อมูล (Filter Logic) const filteredFeed = feed.filter(post => { let pass = true; let postDate = new Date(post.timestamp); let isMyPost = String(post.user_line_id) === String(currentUser.userId); let amITagged = (post.taggedFriends || "").includes(currentUser.userId); // กรองตามประเภท if (filterType === 'related' && !isMyPost && !amITagged) pass = false; if (filterType === 'request') { // โพสต์ที่ต้องการคำยืนยัน (ของฉันที่ยังไม่ผ่าน หรือ ของคนอื่นที่แท็กฉันแล้วยังไม่กด) let isPending = post.status !== 'approved'; if (!(isPending && (isMyPost || amITagged))) pass = false; } // กรองตามวันที่ if (filterDate) { let d = postDate.toISOString().split('T')[0]; if (d !== filterDate) pass = false; } // กรองตามปี if (filterYear) { if (String(postDate.getFullYear()) !== filterYear) pass = false; } return pass; }); if (filteredFeed.length === 0) { container.innerHTML = '
ไม่พบข้อมูลตามเงื่อนไข
'; return; } // 🎨 วาด HTML filteredFeed.forEach(post => { let isMyPost = String(post.user_line_id) === String(currentUser.userId); let taggedIds = (post.taggedFriends && post.taggedFriends.length > 5) ? post.taggedFriends.split(',') : []; let isTeam = taggedIds.length > 0; let verifyList = post.verifies || []; let isVerifiedByMe = verifyList.some(v => String(v.lineId) === String(currentUser.userId)); // 1. 👥 Avatar คนโดนแท็ก let taggedHtml = ''; if (isTeam) { taggedHtml = '
'; taggedIds.forEach(fid => { let fInfo = allUsersMap[fid.trim()]; if(fInfo) taggedHtml += ``; }); taggedHtml += '
'; } // 2. 🔄 แจ้งเตือน Reciprocal (เคยแท็กกันมาก่อน) let reciprocalHtml = ''; if (isTeam && isMyPost) { // เช็คว่าใน taggedIds มีใครเคยแท็กเราไหม (Logic ง่ายๆ ดูจากประวัติเก่าใน Feed นี้แหละ) let history = feed.filter(p => p.id !== post.id && // ไม่ใช่โพสต์ปัจจุบัน String(p.user_line_id) !== String(currentUser.userId) && // ไม่ใช่โพสต์เรา (p.taggedFriends || "").includes(currentUser.userId) && // เขาเคยแท็กเรา taggedIds.includes(String(p.user_line_id)) // และคนนั้นคือคนที่เรากำลังแท็ก ); if (history.length > 0) { reciprocalHtml = `
คุณและ ${history[0].user_name} เคยทำดีร่วมกันล่าสุดเมื่อ ${new Date(history[0].timestamp).toLocaleDateString('th-TH')}
`; } } // 3. 🔘 ปุ่มกด (Logic ใหม่) let btnHtml = ''; if (isMyPost) { // เจ้าของโพสต์: รอเพื่อนยืนยัน (อย่างน้อย 1 คนตามโจทย์) if (isTeam) { let hasWitness = verifyList.length > 0; // แค่ 1 คนก็พอ if (hasWitness || post.status === 'approved') { btnHtml = ` ทีมยืนยันแล้ว (+10)`; } else { btnHtml = ` รอเพื่อนยืนยัน`; } } else { // โพสต์เดี่ยว: รอยืนยันจากใครก็ได้ let vCount = verifyList.length; if (vCount >= 1) btnHtml = `อนุมัติแล้ว (+5)`; else btnHtml = `รอพยาน`; } } else { // คนดูโพสต์ let amITagged = taggedIds.includes(String(currentUser.userId)); if (isTeam) { // ถ้าเป็นโพสต์ทีม คนกดต้องเป็นคนโดนแท็กเท่านั้น if (amITagged) { if (isVerifiedByMe || post.status === 'approved') { btnHtml = ``; } else { btnHtml = ``; } } else { // คนนอกกดไม่ได้ btnHtml = ``; } } else { // โพสต์เดี่ยว คนนอกกดเป็นพยานได้ if (isVerifiedByMe) btnHtml = ``; else btnHtml = ``; } } container.innerHTML += `
${post.user_name}
${new Date(post.timestamp).toLocaleDateString('th-TH', {hour:'2-digit', minute:'2-digit'})} • ${post.virtue} ${taggedHtml}
${btnHtml}
${reciprocalHtml}
${post.note}
${post.image ? `` : ''}
${post.likes ? post.likes.length : 0}
${post.happy==3?'😁':(post.happy==2?'😐':'😞')}
`; }); }) .catch(err => { console.error(err); container.innerHTML = '
โหลดข้อมูลไม่สำเร็จ
'; }); }
